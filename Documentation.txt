# 9. Docker Installation

Remove old conflicting packages if present:

```bash
sudo apt remove -y docker.io docker-doc docker-compose podman-docker containerd runc || true
```

Install Docker repository prerequisites:

```bash
sudo apt update

sudo apt install -y ca-certificates curl
```

Add Docker key:

```bash
sudo install -m 0755 -d /etc/apt/keyrings

sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg \
-o /etc/apt/keyrings/docker.asc

sudo chmod a+r /etc/apt/keyrings/docker.asc
```

Add repository:

```bash
echo \
"deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] \
https://download.docker.com/linux/ubuntu \
$(. /etc/os-release && echo "${UBUNTU_CODENAME:-$VERSION_CODENAME}") stable" | \
sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
```

Install:

```bash
sudo apt update

sudo apt install -y \
docker-ce \
docker-ce-cli \
containerd.io \
docker-buildx-plugin \
docker-compose-plugin
```

Check:

```bash
docker --version
```

Check service:

```bash
sudo systemctl status docker
```

Enable:

```bash
sudo systemctl enable docker
```

Allow Ubuntu user to use Docker:

```bash
sudo usermod -aG docker ubuntu
```

Allow Jenkins to use Docker:

```bash
sudo usermod -aG docker jenkins
```

Restart Jenkins:

```bash
sudo systemctl restart jenkins
```

```yaml
  strategy:

  selector:

  template:

      labels:

    spec:
      containers:
```

# 42. Jenkins CI/CD Pipeline

Complete current Jenkinsfile:

```groovy
pipeline {
    agent any

    triggers {
        githubPush()
    }

    options {
        timestamps()
        disableConcurrentBuilds()
        buildDiscarder(logRotator(numToKeepStr: '10'))
    }

    environment {
        AWS_DEFAULT_REGION = 'ap-south-1'
        AWS_REGION         = 'ap-south-1'
        EKS_CLUSTER        = 'travel-easy-cluster'
        K8S_NAMESPACE      = 'travel-easy'

        DOCKERHUB_USERNAME = 'mokeshambati'
        IMAGE_NAME        = 'travel-easy'
        IMAGE_TAG         = "${BUILD_NUMBER}"
        DOCKER_IMAGE       = "mokeshambati/travel-easy:${BUILD_NUMBER}"

        KUBECONFIG = "${WORKSPACE}/jenkins-kubeconfig"
    }

    stages {

        stage('Checkout') {
            steps {
                checkout scm
            }
        }

        stage('Python Setup') {
            steps {
                sh '''
                    set -e

                    python3 --version

                    python3 -m venv .venv

                    . .venv/bin/activate

                    python -m pip install --upgrade pip

                    pip install -r requirements.txt

                    pip list
                '''
            }
        }

        stage('Django Checks') {
            steps {
                sh '''
                    set -e

                    . .venv/bin/activate

                    python manage.py check

                    python manage.py test
                '''
            }
        }

        stage('SonarQube Analysis') {
            steps {
                script {
                    def scannerHome = tool 'SonarScanner'

                    withSonarQubeEnv('SonarQube') {
                        sh """
                            ${scannerHome}/bin/sonar-scanner \
                              -Dsonar.projectKey=travel-easy \
                              -Dsonar.projectName="Travel Eazy" \
                              -Dsonar.sources=app,project \
                              -Dsonar.python.version=3.11 \
                              -Dsonar.exclusions="**/migrations/**,**/__pycache__/**,**/static/**,**/templates/**"
                        """
                    }
                }
            }
        }

        stage('Trivy Filesystem Scan') {
            steps {
                sh '''
                    echo "===== TRIVY FILESYSTEM SECURITY SCAN ====="

                    trivy fs \
                      --scanners vuln,secret \
                      --severity HIGH,CRITICAL \
                      --exit-code 0 \
                      .
                '''
            }
        }

        stage('Docker Build') {
            steps {
                sh '''
                    set -e

                    docker build \
                      -t ${DOCKER_IMAGE} \
                      .

                    docker images ${DOCKERHUB_USERNAME}/${IMAGE_NAME}
                '''
            }
        }

        stage('Trivy Image Scan') {
            steps {
                sh '''
                    echo "===== TRIVY DOCKER IMAGE SECURITY SCAN ====="

                    trivy image \
                      --severity HIGH,CRITICAL \
                      --exit-code 0 \
                      ${DOCKER_IMAGE}
                '''
            }
        }

        stage('Docker Hub Push') {
            steps {
                withCredentials([
                    usernamePassword(
                        credentialsId: 'dockerhub-creds',
                        usernameVariable: 'DOCKER_USER',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )
                ]) {
                    sh '''
                        set -e

                        echo "$DOCKER_PASSWORD" | \
                        docker login \
                          --username "$DOCKER_USER" \
                          --password-stdin

                        docker push ${DOCKER_IMAGE}

                        docker logout
                    '''
                }
            }
        }

        stage('Configure EKS') {
            steps {
                sh '''
                    set -e

                    aws sts get-caller-identity

                    aws eks update-kubeconfig \
                      --region ${AWS_DEFAULT_REGION} \
                      --name ${EKS_CLUSTER} \
                      --kubeconfig ${KUBECONFIG}

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      get nodes
                '''
            }
        }

        stage('Deploy to EKS') {
            steps {
                sh '''
                    set -e

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      apply \
                      -f k8s/deployment.yaml

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      apply \
                      -f k8s/service.yaml

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      -n ${K8S_NAMESPACE} \
                      set image deployment/travel-easy \
                      travel-easy=${DOCKER_IMAGE}

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      -n ${K8S_NAMESPACE} \
                      rollout status deployment/travel-easy \
                      --timeout=5m
                '''
            }
        }

        stage('Verify Deployment') {
            steps {
                sh '''
                    set -e

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      get deployment \
                      -n ${K8S_NAMESPACE}

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      get pods \
                      -n ${K8S_NAMESPACE} \
                      -o wide

                    kubectl \
                      --kubeconfig ${KUBECONFIG} \
                      get svc \
                      -n ${K8S_NAMESPACE}
                '''
            }
        }
    }

    post {

        success {
            echo """
            ========================================
            TRAVEL EAZY CI/CD PIPELINE SUCCESSFUL
            ========================================
            Docker Image : ${DOCKER_IMAGE}
            EKS Cluster  : ${EKS_CLUSTER}
            Namespace    : ${K8S_NAMESPACE}
            ========================================
            """
        }

        failure {
            echo """
            ========================================
            TRAVEL EAZY CI/CD PIPELINE FAILED
            ========================================
            Check the failed stage in Jenkins Console Output.
            ========================================
            """
        }

        always {
            sh '''
                rm -f "${KUBECONFIG}" || true
            '''
        }
    }
}
```

# 69. Creating travel-easy-alerts.yaml Using nano

**WHERE: Ubuntu EC2 terminal**

```bash
cd ~
```

Create:

```bash
nano travel-easy-alerts.yaml
```

Paste:

```yaml
apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: travel-easy-alerts
  namespace: monitoring
  labels:
    release: monitoring
spec:
  groups:
    - name: travel-easy.rules
      rules:

        - alert: TravelEasyPodDown
          expr: kube_deployment_status_replicas_available{namespace="travel-easy",deployment="travel-easy"} < 1
          for: 2m
          labels:
            severity: critical
          annotations:
            summary: "Travel Eazy application is down"
            description: "No Travel Eazy replica is currently available."

        - alert: TravelEasyHighCPU
          expr: sum(rate(container_cpu_usage_seconds_total{namespace="travel-easy",container!="",container!="POD"}[5m])) > 0.8
          for: 5m
          labels:
            severity: warning
          annotations:
            summary: "Travel Eazy CPU usage is high"
            description: "Travel Eazy CPU usage has exceeded 0.8 CPU cores for 5 minutes."

        - alert: TravelEasyPodRestarting
          expr: increase(kube_pod_container_status_restarts_total{namespace="travel-easy"}[10m]) > 2
          for: 2m
          labels:
            severity: warning
          annotations:
            summary: "Travel Eazy pod is restarting"
            description: "A Travel Eazy container has restarted multiple times."
```

Save:

```text
CTRL + O
ENTER
CTRL + X
```

Check:

```bash
cat travel-easy-alerts.yaml
```

# 72. Creating AlertmanagerConfig YAML Using nano

Create using:

```bash
nano YOUR-ACTUAL-ALERTMANAGER-CONFIG-FILENAME.yaml
```

```yaml
apiVersion: monitoring.coreos.com/v1alpha1
kind: AlertmanagerConfig
metadata:
  name: travel-easy-email
  namespace: travel-easy
  labels:
    release: monitoring
spec:
  route:
    receiver: travel-easy-email
    matchers:
      - name: alertname
        matchType: =~
        value: "TravelEasy.*"

  receivers:
    - name: travel-easy-email
      emailConfigs:
        - to: "YOUR_GMAIL_ADDRESS"
          from: "YOUR_GMAIL_ADDRESS"
          smarthost: "smtp.gmail.com:587"
          authUsername: "YOUR_GMAIL_ADDRESS"
          authPassword:
            name: alertmanager-smtp
            key: smtp-password
          requireTLS: true
          sendResolved: true
```

# 132. Resume Project Description

**Travel Eazy — Cloud-Native CI/CD Deployment and Monitoring**

* Containerized a Django-based Travel Eazy application using Docker and deployed it on Amazon EKS.
* Implemented an automated Jenkins CI/CD pipeline integrated with GitHub for source checkout, Django validation, Docker image creation, Docker Hub publishing, and Kubernetes deployment.
* Integrated SonarQube for automated code-quality analysis and Trivy for filesystem and Docker image security scanning.
* Configured Amazon EKS deployment and service manifests for Kubernetes-based application delivery.
* Implemented Prometheus and Grafana monitoring with application, pod, and node-level metrics.
* Configured PrometheusRule and Alertmanager for application availability, CPU, and pod-restart alerts with Gmail notifications.
* Used AWS IAM role-based authentication for Jenkins-to-EKS access without storing long-lived AWS credentials.

# 133. Interview Explanation

## Project Introduction

```text
I developed a Django-based Travel Eazy application and implemented a complete
CI/CD workflow using GitHub, Jenkins, Docker, SonarQube, Trivy, Docker Hub,
and Amazon EKS.
```

## CI/CD

```text
Whenever code is pushed to GitHub, a webhook triggers Jenkins.
Jenkins checks out the code, installs Python dependencies, runs Django checks
and tests, performs SonarQube analysis, scans the source using Trivy, builds
the Docker image, scans the image, pushes it to Docker Hub, and deploys the
image to Amazon EKS.
```

## Kubernetes

```text
The application runs inside a Kubernetes Deployment in the travel-easy
namespace. A LoadBalancer Service exposes port 80 and forwards traffic to
the application container on port 8000.
```

## Monitoring

```text
Prometheus collects Kubernetes and application-related metrics.
Grafana displays the metrics through dashboards.
Alertmanager handles alerts generated by Prometheus and sends email
notifications through Gmail SMTP.
```
